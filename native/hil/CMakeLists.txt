cmake_minimum_required(VERSION 3.24)
project(monaka_openvr_hil LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This HIL helper targets Windows x64 only")
endif()
set(OPENVR_SOURCE "" CACHE PATH "Clean pinned OpenVR checkout (read only)")
find_package(Git REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${OPENVR_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE openvr_head OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT openvr_head STREQUAL "91825305130f446f82054c1ec3d416321ace0072")
    message(FATAL_ERROR "OpenVR pin mismatch")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${OPENVR_SOURCE}" status --porcelain
    OUTPUT_VARIABLE openvr_dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(openvr_dirty)
    message(FATAL_ERROR "OpenVR checkout must be clean")
endif()
add_executable(monaka_openvr_hil_action main.cpp)
target_compile_features(monaka_openvr_hil_action PRIVATE cxx_std_20)
target_include_directories(monaka_openvr_hil_action PRIVATE "${OPENVR_SOURCE}/headers")
target_link_libraries(monaka_openvr_hil_action PRIVATE "${OPENVR_SOURCE}/lib/win64/openvr_api.lib" ole32)
add_custom_command(TARGET monaka_openvr_hil_action POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${OPENVR_SOURCE}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:monaka_openvr_hil_action>")
add_executable(monaka_hil_action_test action_test.cpp)
target_compile_features(monaka_hil_action_test PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(monaka_openvr_hil_action PRIVATE /W4 /WX /EHsc)
    target_compile_options(monaka_hil_action_test PRIVATE /W4 /WX /EHsc)
endif()
enable_testing()
add_test(NAME hil_action_logic COMMAND monaka_hil_action_test)
add_test(NAME hil_action_help COMMAND monaka_openvr_hil_action --help)
set_tests_properties(hil_action_help PROPERTIES PASS_REGULAR_EXPRESSION "Markers are action observations")
# Runtime-independent integration checks; these commands are rejected before VR_Init.
add_test(NAME hil_action_invalid COMMAND monaka_openvr_hil_action --action invalid)
add_test(NAME hil_action_seated_guard COMMAND monaka_openvr_hil_action --action seated-reset)
add_test(NAME hil_action_standing_guard COMMAND monaka_openvr_hil_action --action standing-reset)
set_tests_properties(hil_action_invalid hil_action_seated_guard hil_action_standing_guard PROPERTIES WILL_FAIL TRUE)
add_test(NAME hil_action_cli_records
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_cli.py" "$<TARGET_FILE:monaka_openvr_hil_action>")
